18real gamma_lcdf_logx(real log_x, real a) {
19 if (!(a > 0) || is_inf(a)) {
20 reject("gamma_lcdf_logx: shape must be positive finite, found ", a);
21 }
22 real x = exp(log_x);
23 if (!(x < 0.5 * (a + 1))) {
24 return gamma_lcdf(x | a, 1);
25 }
26 // Each term is at most half the one before
27 real term = 1;
28 real total = 1;
29 real n = a;
30 while (term > 1e-16 * total) {
31 n += 1;
32 term *= x / n;
33 total += term;
34 }
35 return a * log_x - x - lgamma(a + 1) + log(total);
36}

165real dist_lcdf(real delay, array[] real params, int dist_id) {
166 if (dist_has_positive_support(dist_id) && delay <= 0) {
167 return negative_infinity();
168 }
169
170 // IDs match pcd_distributions$stan_id in R
171 // Guarded so a lower-tail underflow cannot put a NaN partial on the tape.
172 // The downstream `exp(-inf)` differentiates to 0.
173 if (dist_id == 1) {
174 return lognormal_lcdf_underflows(delay, params[1], params[2])
175 ? negative_infinity()
176 : lognormal_lcdf(delay | params[1], params[2]);
177 }
178 else if (dist_id == 2) {
179 return gamma_lcdf_logx(log(delay) + gamma_log_rate(params[2]), params[1]);
180 }
181 else if (dist_id == 3) return weibull_lcdf(delay | params[1], params[2]);
182 else if (dist_id == 4) return exponential_lcdf(delay | params[1]);
183 else if (dist_id == 5) return gengamma_lcdf(delay | params[1], params[2], params[3]);
184 else if (dist_id == 9) {
185 // The Beta has support on [0, 1], and beta_lcdf rejects delay > 1
186 return delay >= 1 ? 0 : beta_lcdf(delay | params[1], params[2]);
187 }
188 else if (dist_id == 12) return cauchy_lcdf(delay | params[1], params[2]);
189 else if (dist_id == 13) return chi_square_lcdf(delay | params[1]);
190 else if (dist_id == 15) return gumbel_lcdf(delay | params[1], params[2]);
191 else if (dist_id == 16) return inv_gamma_lcdf(delay | params[1], params[2]);
192 else if (dist_id == 17) return logistic_lcdf(delay | params[1], params[2]);
193 else if (dist_id == 18) return normal_lcdf(delay | params[1], params[2]);
194 else if (dist_id == 19) return inv_chi_square_lcdf(delay | params[1]);
195 else if (dist_id == 20) return double_exponential_lcdf(delay | params[1], params[2]);
196 else if (dist_id == 21) return pareto_lcdf(delay | params[1], params[2]);
197 else if (dist_id == 22) return scaled_inv_chi_square_lcdf(delay | params[1], params[2]);
198 else if (dist_id == 23) return student_t_lcdf(delay | params[1], params[2], params[3]);
199 else if (dist_id == 24) return uniform_lcdf(delay | params[1], params[2]);
200 else if (dist_id == 25) return von_mises_lcdf(delay | params[1], params[2]);
201 else if (dist_id == 26) {
202 // Non-parametric step: params = [boundaries (K+1), pmf (K)].
203 int K = (size(params) - 1) %/% 2;
204 return pstep_lcdf(
205 delay | to_vector(segment(params, 1, K + 1)),
206 to_vector(segment(params, K + 2, K))
207 );
208 }
209 else if (dist_id == 27 || dist_id == 28) {
210 // Non-parametric discrete hazard: params = [boundaries (K+1),
211 // hazards (K)] with hazards[K] = 1. RW (27) and RE (28) share the
212 // same likelihood; they only differ in the prior.
213 int K = (size(params) - 1) %/% 2;
214 return phazard_lcdf(
215 delay | to_vector(segment(params, 1, K + 1)),
216 to_vector(segment(params, K + 2, K))
217 );
218 }
219 else reject("Invalid distribution identifier: ", dist_id);
220}
