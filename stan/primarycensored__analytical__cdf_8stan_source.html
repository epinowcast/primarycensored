42real primarycensored_gamma_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
43 real shape = params[1];
44 real rate = params[2];
45 real log_window = log(pwindow);
46 // log E where E = k * theta = shape / rate is the mean of the delay
47 real log_E = log(shape) - log(rate);
48
49 // F_T(d; k) and the recursion to F_T(d; k+1):
50 // P(k+1, y) = P(k, y) - y^k e^{-y} / Gamma(k+1), with y = rate * d
51 real log_F_T_d_k = gamma_lcdf(d | shape, rate);
52 real gamma_kp1_pdf_log_d
53 = shape * log(rate * d) - rate * d - lgamma(shape + 1);
54 real log_F_T_d_kp1 = log_diff_exp(log_F_T_d_k, gamma_kp1_pdf_log_d);
55
56 // q-dependent terms. Final algebra is unified; only a guard to avoid
57 // log_diff_exp(-inf, -inf) and log(0) when q == 0 (q is data, so autodiff
58 // is unaffected by this branch).
59 real log_q_F_T_q; // log(q * F_T(q; k))
60 real log_E_tF_T_q; // log(E * F_T(q; k+1))
61 if (q > 0) {
62 real log_F_T_q_k = gamma_lcdf(q | shape, rate);
63 real gamma_kp1_pdf_log_q
64 = shape * log(rate * q) - rate * q - lgamma(shape + 1);
65 real log_F_T_q_kp1 = log_diff_exp(log_F_T_q_k, gamma_kp1_pdf_log_q);
66 log_q_F_T_q = log(q) + log_F_T_q_k;
67 log_E_tF_T_q = log_E + log_F_T_q_kp1;
68 } else {
69 log_q_F_T_q = negative_infinity();
70 log_E_tF_T_q = negative_infinity();
71 }
72
73 // Unified form: F_{S+}(d) = (A - B) / w_P with A, B sums of positives:
74 // A = d * F_T(d; k) + E * F_T(q; k+1)
75 // B = q * F_T(q; k) + E * F_T(d; k+1)
76 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
77 real log_A = log_sum_exp(log(d) + log_F_T_d_k, log_E_tF_T_q);
78 real log_B = log_sum_exp(log_q_F_T_q, log_E + log_F_T_d_kp1);
79
80 return log_diff_exp(log_A, log_B) - log_window;
81}

95real primarycensored_lognormal_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
96 real mu = params[1];
97 real sigma = params[2];
98 real mu_sigma2 = mu + square(sigma);
99 real log_window = log(pwindow);
100 // log E where E = exp(mu + sigma^2/2) is the mean of the delay
101 real log_E = mu + 0.5 * square(sigma);
102
103 real log_F_T_d = lognormal_lcdf(d | mu, sigma);
104 real log_tF_T_d = lognormal_lcdf(d | mu_sigma2, sigma);
105
106 // q-dependent terms (guard only to avoid log(0); final algebra is unified).
107 real log_q_F_T_q; // log(q * F_T(q))
108 real log_E_tF_T_q; // log(E * tilde F_T(q))
109 if (q > 0) {
110 real log_F_T_q = lognormal_lcdf(q | mu, sigma);
111 real log_tF_T_q = lognormal_lcdf(q | mu_sigma2, sigma);
112 log_q_F_T_q = log(q) + log_F_T_q;
113 log_E_tF_T_q = log_E + log_tF_T_q;
114 } else {
115 log_q_F_T_q = negative_infinity();
116 log_E_tF_T_q = negative_infinity();
117 }
118
119 // Unified form: F_{S+}(d) = (A - B) / w_P with
120 // A = d * F_T(d) + E * tilde F_T(q)
121 // B = q * F_T(q) + E * tilde F_T(d)
122 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
123 real log_A = log_sum_exp(log(d) + log_F_T_d, log_E_tF_T_q);
124 real log_B = log_sum_exp(log_q_F_T_q, log_E + log_tF_T_d);
125
126 return log_diff_exp(log_A, log_B) - log_window;
127}

161real primarycensored_weibull_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
162 real shape = params[1];
163 real scale = params[2];
164 real log_window = log(pwindow);
165 real log_scale = log(scale);
166
167 // For Weibull: E = scale (lambda) and tilde F_T(t) = g(t; lambda, k), so
168 // log(E * tilde F_T(t)) = log(scale) + log_weibull_g(t, shape, scale).
169 real log_F_T_d = weibull_lcdf(d | shape, scale);
170 real log_E_tF_T_d = log_scale + log_weibull_g(d, shape, scale);
171
172 // q-dependent terms (guard only to avoid log(0); final algebra is unified).
173 real log_q_F_T_q; // log(q * F_T(q))
174 real log_E_tF_T_q; // log(E * tilde F_T(q)) = log(scale * g(q; lambda, k))
175 if (q > 0) {
176 log_q_F_T_q = log(q) + weibull_lcdf(q | shape, scale);
177 log_E_tF_T_q = log_scale + log_weibull_g(q, shape, scale);
178 } else {
179 log_q_F_T_q = negative_infinity();
180 log_E_tF_T_q = negative_infinity();
181 }
182
183 // Unified form: F_{S+}(d) = (A - B) / w_P with
184 // A = d * F_T(d) + scale * g(q; lambda, k)
185 // B = q * F_T(q) + scale * g(d; lambda, k)
186 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
187 real log_A = log_sum_exp(log(d) + log_F_T_d, log_E_tF_T_q);
188 real log_B = log_sum_exp(log_q_F_T_q, log_E_tF_T_d);
189
190 return log_diff_exp(log_A, log_B) - log_window;
191}

212real primarycensored_gengamma_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
213 real shape = params[1];
214 real scale = params[2];
215 real k = params[3];
216 real k_shift = k + inv(shape);
217 real log_window = log(pwindow);
218 // log E where E = scale * Gamma(k + 1/shape) / Gamma(k) is the mean of the

220 real log_E = log(scale) + lgamma(k_shift) - lgamma(k);
221
222 real log_F_T_d = gengamma_lcdf(d | shape, scale, k);
223 real log_tF_T_d = gengamma_lcdf(d | shape, scale, k_shift);

229 log_q_F_T_q = log(q) + gengamma_lcdf(q | shape, scale, k);
230 log_E_tF_T_q = log_E + gengamma_lcdf(q | shape, scale, k_shift);
231 } else {
232 log_q_F_T_q = negative_infinity();
233 log_E_tF_T_q = negative_infinity();

251real primarycensored_analytical_lcdf_raw(data real d, int dist_id,
252 array[] real params,
253 data real pwindow,
254 int primary_id,
255 array[] real primary_params) {
256 real q = max({d - pwindow, 0});
257
258 if (dist_id == 2 && primary_id == 1) {
259 return primarycensored_gamma_uniform_lcdf(d | q, params, pwindow);
260 } else if (dist_id == 1 && primary_id == 1) {
261 return primarycensored_lognormal_uniform_lcdf(d | q, params, pwindow);
262 } else if (dist_id == 3 && primary_id == 1) {
263 return primarycensored_weibull_uniform_lcdf(d | q, params, pwindow);
264 } else if (dist_id == 5 && primary_id == 1) {
265 return primarycensored_gengamma_uniform_lcdf(d | q, params, pwindow);
266 } else if (dist_id == 26) {
267 // params = [boundaries (K+1), pmf (K)]; length 2*K + 1.
268 int K = (size(params) - 1) %/% 2;
269 return discretestep_lcdf(
270 d | to_vector(segment(params, 1, K + 1)),
271 to_vector(segment(params, K + 2, K)),
272 primary_id, primary_params, pwindow
273 );
274 } else if (dist_id == 27 || dist_id == 28) {
275 // params = [boundaries (K+1), hazards (K)]; length 2*K + 1. The last
276 // hazard must equal 1. RW (27) and RE (28) only differ in their
277 // prior so they share this likelihood dispatch.
278 int K = (size(params) - 1) %/% 2;
279 return discretehazard_lcdf(
280 d | to_vector(segment(params, 1, K + 1)),
281 to_vector(segment(params, K + 2, K)),
282 primary_id, primary_params, pwindow
283 );
284 }
285 return negative_infinity();
286}
