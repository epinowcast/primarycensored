68real primarycensored_uniform_lcdf_from_terms(vector terms_d, vector terms_q,
69 data real pwindow) {
70 real log_A = log_sum_exp(terms_d[1], terms_q[2]);
71 real log_B = log_sum_exp(terms_q[1], terms_d[2]);
72 // Deep enough into the lower tail every term underflows together. Both
73 // are then `-inf` and `log_diff_exp` would give NaN, so return the limit
74 // directly.
75 if (log_A == negative_infinity() && log_B == negative_infinity()) {
76 return negative_infinity();
77 }
78 return log_diff_exp(log_A, log_B) - log(pwindow);
79}

91vector primarycensored_gamma_uniform_terms(real t,
92 array[] real params) {
93 if (t <= 0) {
94 return rep_vector(negative_infinity(), 2);
95 }
96 real shape = params[1];
97 real rate = params[2];
98 // log E where E = k * theta = shape / rate is the mean of the delay
99 real log_E = log(shape) - log(rate);
100 // F_T(t; k) and the recursion to F_T(t; k+1):
101 // P(k+1, y) = P(k, y) - y^k e^{-y} / Gamma(k+1), with y = rate * t
102 real log_F_T_k = gamma_lcdf(t | shape, rate);
103 real gamma_kp1_pdf_log = shape * log(rate * t) - rate * t
104 - lgamma(shape + 1);
105 real log_F_T_kp1 = log_diff_exp(log_F_T_k, gamma_kp1_pdf_log);
106 return [log(t) + log_F_T_k, log_E + log_F_T_kp1]';
107}

123vector primarycensored_lognormal_uniform_terms(real t,
124 array[] real params) {
125 real mu = params[1];
126 real sigma = params[2];
127 real mu_sigma2 = mu + square(sigma);
128 // log E where E = exp(mu + sigma^2/2) is the mean of the delay
129 real log_E = mu + 0.5 * square(sigma);
130 real log_t_F_T = lognormal_lcdf_underflows(t, mu, sigma)
131 ? negative_infinity()
132 : log(t) + lognormal_lcdf(t | mu, sigma);
133 real log_E_tF_T = lognormal_lcdf_underflows(t, mu_sigma2, sigma)
134 ? negative_infinity()
135 : log_E + lognormal_lcdf(t | mu_sigma2, sigma);
136 return [log_t_F_T, log_E_tF_T]';
137}

153real log_weibull_g(real t, real shape, real scale) {
154 real x = pow(t * inv(scale), shape);
155 real a = 1 + inv(shape);
156 // gamma_lcdf(x | a, 1) is log(gamma_p(a, x)), but reverse-mode gamma_p()
157 // returns zero gradients for x / a > 10
158 // (https://github.com/stan-dev/math/issues/2006). gamma_lcdf() has the
159 // same value and computes its own gradients without that cutoff.
160 return gamma_lcdf(x | a, 1) + lgamma(a);
161}

205vector primarycensored_gengamma_uniform_terms(real t,
206 array[] real params) {
207 if (t <= 0) {
208 return rep_vector(negative_infinity(), 2);
209 }
210 real shape = params[1];
211 real scale = params[2];
212 real k = params[3];
213 real k_shift = k + inv(shape);
214 real log_E = log(scale) + lgamma(k_shift) - lgamma(k);
215 return [
216 log(t) + gengamma_lcdf(t | shape, scale, k),
217 log_E + gengamma_lcdf(t | shape, scale, k_shift)
218 ]';
219}

337real primarycensored_analytical_lcdf_raw(data real d, int dist_id,
338 array[] real params,
339 data real pwindow,
340 int primary_id,
341 array[] real primary_params) {
342 real q = max({d - pwindow, 0});
343
344 if (dist_id == 2 && primary_id == 1) {
345 return primarycensored_gamma_uniform_lcdf(d | q, params, pwindow);
346 } else if (dist_id == 1 && primary_id == 1) {
347 return primarycensored_lognormal_uniform_lcdf(d | q, params, pwindow);
348 } else if (dist_id == 3 && primary_id == 1) {
349 return primarycensored_weibull_uniform_lcdf(d | q, params, pwindow);
350 } else if (dist_id == 5 && primary_id == 1) {
351 return primarycensored_gengamma_uniform_lcdf(d | q, params, pwindow);
352 } else if (dist_id == 26) {
353 // params = [boundaries (K+1), pmf (K)]; length 2*K + 1.
354 int K = (size(params) - 1) %/% 2;
355 return discretestep_lcdf(
356 d | to_vector(segment(params, 1, K + 1)),
357 to_vector(segment(params, K + 2, K)),
358 primary_id, primary_params, pwindow
359 );
360 } else if (dist_id == 27 || dist_id == 28) {
361 // params = [boundaries (K+1), hazards (K)]; length 2*K + 1. The last
362 // hazard must equal 1. RW (27) and RE (28) only differ in their
363 // prior so they share this likelihood dispatch.
364 int K = (size(params) - 1) %/% 2;
365 return discretehazard_lcdf(
366 d | to_vector(segment(params, 1, K + 1)),
367 to_vector(segment(params, K + 2, K)),
368 primary_id, primary_params, pwindow
369 );
370 }
371 return negative_infinity();
372}

390real primarycensored_analytical_lcdf(data real d, int dist_id,
391 array[] real params,
392 data real pwindow, data real L,
393 data real D, int primary_id,
394 array[] real primary_params) {
395 if (d <= L) return negative_infinity();
396 if (d >= D) return 0;
397

399 d, dist_id, params, pwindow, primary_id, primary_params
400 );
401

404 vector[2] bounds = primarycensored_truncation_bounds(
405 L, D, dist_id, params, pwindow, primary_id, primary_params

408 real log_cdf_D = bounds[2];
409

411 result = primarycensored_apply_truncation(result, log_cdf_L, log_normalizer, L);
412 }
413
414 return result;
415}
