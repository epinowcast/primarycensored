68real primarycensored_uniform_lcdf_from_terms(vector terms_d, vector terms_q,
69 data real pwindow) {
70 real log_A = log_sum_exp(terms_d[1], terms_q[2]);
71 real log_B = log_sum_exp(terms_q[1], terms_d[2]);
72 // Deep enough into the lower tail every term underflows together. Both
73 // are then `-inf` and `log_diff_exp` would give NaN, so return the limit
74 // directly.
75 if (log_A == negative_infinity() && log_B == negative_infinity()) {
76 return negative_infinity();
77 }
78 return log_diff_exp(log_A, log_B) - log(pwindow);
79}

96vector primarycensored_gamma_uniform_terms(real t,
97 array[] real params) {
98 if (t <= 0) {
99 return rep_vector(negative_infinity(), 2);
100 }
101 real shape = params[1];
102 real rate = params[2];
103 real log_rate = gamma_log_rate(rate);
104 // log E where E = k * theta = shape / rate is the mean of the delay
105 real log_E = log(shape) - log_rate;
106 real log_t = log(t);
107 real log_y = log_t + log_rate;
108 real log_F_T_kp1 = gamma_lcdf_logx(log_y, shape + 1);
109 real log_lead = shape * log_y - exp(log_y) - lgamma(shape + 1);
110 return [
111 log_t + log_sum_exp(log_F_T_kp1, log_lead),
112 log_E + log_F_T_kp1
113 ]';
114}

130vector primarycensored_lognormal_uniform_terms(real t,
131 array[] real params) {
132 real mu = params[1];
133 real sigma = params[2];
134 real mu_sigma2 = mu + square(sigma);
135 // log E where E = exp(mu + sigma^2/2) is the mean of the delay
136 real log_E = mu + 0.5 * square(sigma);
137 real log_t_F_T = lognormal_lcdf_underflows(t, mu, sigma)
138 ? negative_infinity()
139 : log(t) + lognormal_lcdf(t | mu, sigma);
140 real log_E_tF_T = lognormal_lcdf_underflows(t, mu_sigma2, sigma)
141 ? negative_infinity()
142 : log_E + lognormal_lcdf(t | mu_sigma2, sigma);
143 return [log_t_F_T, log_E_tF_T]';
144}

160real log_weibull_g(real t, real shape, real scale) {
161 real x = pow(t * inv(scale), shape);
162 real a = 1 + inv(shape);
163 // gamma_lcdf(x | a, 1) is log(gamma_p(a, x)), but reverse-mode gamma_p()
164 // returns zero gradients for x / a > 10
165 // (https://github.com/stan-dev/math/issues/2006). gamma_lcdf() has the
166 // same value and computes its own gradients without that cutoff.
167 return gamma_lcdf(x | a, 1) + lgamma(a);
168}

212vector primarycensored_gengamma_uniform_terms(real t,
213 array[] real params) {
214 if (t <= 0) {
215 return rep_vector(negative_infinity(), 2);
216 }
217 real shape = params[1];
218 real scale = params[2];
219 real k = params[3];
220 real k_shift = k + inv(shape);
221 real log_E = log(scale) + lgamma(k_shift) - lgamma(k);
222 return [
223 log(t) + gengamma_lcdf(t | shape, scale, k),
224 log_E + gengamma_lcdf(t | shape, scale, k_shift)

344real primarycensored_analytical_lcdf_raw(data real d, int dist_id,
345 array[] real params,
346 data real pwindow,
347 int primary_id,
348 array[] real primary_params) {
349 real q = max({d - pwindow, 0});
350
351 if (dist_id == 2 && primary_id == 1) {
352 return primarycensored_gamma_uniform_lcdf(d | q, params, pwindow);
353 } else if (dist_id == 1 && primary_id == 1) {
354 return primarycensored_lognormal_uniform_lcdf(d | q, params, pwindow);
355 } else if (dist_id == 3 && primary_id == 1) {
356 return primarycensored_weibull_uniform_lcdf(d | q, params, pwindow);
357 } else if (dist_id == 5 && primary_id == 1) {
358 return primarycensored_gengamma_uniform_lcdf(d | q, params, pwindow);
359 } else if (dist_id == 26) {
360 // params = [boundaries (K+1), pmf (K)]; length 2*K + 1.
361 int K = (size(params) - 1) %/% 2;
362 return discretestep_lcdf(
363 d | to_vector(segment(params, 1, K + 1)),
364 to_vector(segment(params, K + 2, K)),
365 primary_id, primary_params, pwindow
366 );
367 } else if (dist_id == 27 || dist_id == 28) {
368 // params = [boundaries (K+1), hazards (K)]; length 2*K + 1. The last
369 // hazard must equal 1. RW (27) and RE (28) only differ in their
370 // prior so they share this likelihood dispatch.
371 int K = (size(params) - 1) %/% 2;
372 return discretehazard_lcdf(
373 d | to_vector(segment(params, 1, K + 1)),
374 to_vector(segment(params, K + 2, K)),
375 primary_id, primary_params, pwindow
376 );
377 }
378 return negative_infinity();
379}

397real primarycensored_analytical_lcdf(data real d, int dist_id,
398 array[] real params,
399 data real pwindow, data real L,
400 data real D, int primary_id,
401 array[] real primary_params) {
402 if (d <= L) return negative_infinity();
403 if (d >= D) return 0;
404
405 real result = primarycensored_analytical_lcdf_raw(
406 d, dist_id, params, pwindow, primary_id, primary_params
407 );
408
409 // Apply truncation normalization
410 if (!is_inf(D) || L > 0) {
411 vector[2] bounds = primarycensored_truncation_bounds(
412 L, D, dist_id, params, pwindow, primary_id, primary_params
413 );
414 real log_cdf_L = bounds[1];
415 real log_cdf_D = bounds[2];
416
417 real log_normalizer = primarycensored_log_normalizer(log_cdf_D, log_cdf_L, L);
418 result = primarycensored_apply_truncation(result, log_cdf_L, log_normalizer, L);
419 }
420
421 return result;
422}
