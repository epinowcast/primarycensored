42real primarycensored_gamma_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
43 real shape = params[1];
44 real rate = params[2];
45 real log_window = log(pwindow);
46 // log E where E = k * theta = shape / rate is the mean of the delay
47 real log_E = log(shape) - log(rate);
48
49 // F_T(d; k) and the recursion to F_T(d; k+1):
50 // P(k+1, y) = P(k, y) - y^k e^{-y} / Gamma(k+1), with y = rate * d
51 real log_F_T_d_k = gamma_lcdf(d | shape, rate);
52 real gamma_kp1_pdf_log_d
53 = shape * log(rate * d) - rate * d - lgamma(shape + 1);
54 real log_F_T_d_kp1 = log_diff_exp(log_F_T_d_k, gamma_kp1_pdf_log_d);
55
56 // q-dependent terms. Final algebra is unified; only a guard to avoid
57 // log_diff_exp(-inf, -inf) and log(0) when q == 0 (q is data, so autodiff
58 // is unaffected by this branch).
59 real log_q_F_T_q; // log(q * F_T(q; k))
60 real log_E_tF_T_q; // log(E * F_T(q; k+1))
61 if (q > 0) {
62 real log_F_T_q_k = gamma_lcdf(q | shape, rate);
63 real gamma_kp1_pdf_log_q
64 = shape * log(rate * q) - rate * q - lgamma(shape + 1);
65 real log_F_T_q_kp1 = log_diff_exp(log_F_T_q_k, gamma_kp1_pdf_log_q);
66 log_q_F_T_q = log(q) + log_F_T_q_k;
67 log_E_tF_T_q = log_E + log_F_T_q_kp1;
68 } else {
69 log_q_F_T_q = negative_infinity();
70 log_E_tF_T_q = negative_infinity();
71 }
72
73 // Unified form: F_{S+}(d) = (A - B) / w_P with A, B sums of positives:
74 // A = d * F_T(d; k) + E * F_T(q; k+1)
75 // B = q * F_T(q; k) + E * F_T(d; k+1)
76 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
77 real log_A = log_sum_exp(log(d) + log_F_T_d_k, log_E_tF_T_q);
78 real log_B = log_sum_exp(log_q_F_T_q, log_E + log_F_T_d_kp1);
79
80 return log_diff_exp(log_A, log_B) - log_window;
81}

95real primarycensored_lognormal_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
96 real mu = params[1];
97 real sigma = params[2];
98 real mu_sigma2 = mu + square(sigma);
99 real log_window = log(pwindow);
100 // log E where E = exp(mu + sigma^2/2) is the mean of the delay
101 real log_E = mu + 0.5 * square(sigma);
102
103 // Each term is formed whole and dropped whole. Adding a `-inf` log CDF to
104 // the parameter-dependent `log(d)` or `log_E` first would leave an edge
105 // back to the parameters that `log_sum_exp` differentiates to
106 // `exp(-inf - -inf)`. `q <= 0` underflows on the same test, so it needs no
107 // separate branch.
108 real log_d_F_T_d = lognormal_lcdf_underflows(d, mu, sigma)
109 ? negative_infinity()
110 : log(d) + lognormal_lcdf(d | mu, sigma);
111 real log_E_tF_T_d = lognormal_lcdf_underflows(d, mu_sigma2, sigma)
112 ? negative_infinity()
113 : log_E + lognormal_lcdf(d | mu_sigma2, sigma);
114 real log_q_F_T_q = lognormal_lcdf_underflows(q, mu, sigma)
115 ? negative_infinity()
116 : log(q) + lognormal_lcdf(q | mu, sigma);
117 real log_E_tF_T_q = lognormal_lcdf_underflows(q, mu_sigma2, sigma)
118 ? negative_infinity()
119 : log_E + lognormal_lcdf(q | mu_sigma2, sigma);
120
121 // Unified form: F_{S+}(d) = (A - B) / w_P with
122 // A = d * F_T(d) + E * tilde F_T(q)
123 // B = q * F_T(q) + E * tilde F_T(d)
124 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
125 real log_A = log_sum_exp(log_d_F_T_d, log_E_tF_T_q);
126 real log_B = log_sum_exp(log_q_F_T_q, log_E_tF_T_d);
127
128 // Deep enough into the lower tail every term underflows together. Both
129 // are then constant `-inf` and `log_diff_exp` would give NaN, so return
130 // the limit directly.
131 if (is_inf(log_A)) {
132 return negative_infinity();
133 }
134
135 return log_diff_exp(log_A, log_B) - log_window;
136}

152real log_weibull_g(real t, real shape, real scale) {
153 real x = pow(t * inv(scale), shape);
154 real a = 1 + inv(shape);
155 return log(gamma_p(a, x)) + lgamma(a);

170real primarycensored_weibull_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
171 real shape = params[1];
172 real scale = params[2];
173 real log_window = log(pwindow);
174 real log_scale = log(scale);
175
176 // For Weibull: E = scale (lambda) and tilde F_T(t) = g(t; lambda, k), so
177 // log(E * tilde F_T(t)) = log(scale) + log_weibull_g(t, shape, scale).
178 real log_F_T_d = weibull_lcdf(d | shape, scale);
179 real log_E_tF_T_d = log_scale + log_weibull_g(d, shape, scale);
180
181 // q-dependent terms (guard only to avoid log(0); final algebra is unified).
182 real log_q_F_T_q; // log(q * F_T(q))
183 real log_E_tF_T_q; // log(E * tilde F_T(q)) = log(scale * g(q; lambda, k))
184 if (q > 0) {
185 log_q_F_T_q = log(q) + weibull_lcdf(q | shape, scale);
186 log_E_tF_T_q = log_scale + log_weibull_g(q, shape, scale);
187 } else {
188 log_q_F_T_q = negative_infinity();
189 log_E_tF_T_q = negative_infinity();
190 }
191
192 // Unified form: F_{S+}(d) = (A - B) / w_P with
193 // A = d * F_T(d) + scale * g(q; lambda, k)
194 // B = q * F_T(q) + scale * g(d; lambda, k)
195 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
196 real log_A = log_sum_exp(log(d) + log_F_T_d, log_E_tF_T_q);
197 real log_B = log_sum_exp(log_q_F_T_q, log_E_tF_T_d);
198
199 return log_diff_exp(log_A, log_B) - log_window;
200}

221real primarycensored_gengamma_uniform_lcdf(data real d, real q, array[] real params, data real pwindow) {
222 real shape = params[1];
223 real scale = params[2];
224 real k = params[3];
225 real k_shift = k + inv(shape);
226 real log_window = log(pwindow);
227 // log E where E = scale * Gamma(k + 1/shape) / Gamma(k) is the mean of the
228 // delay
229 real log_E = log(scale) + lgamma(k_shift) - lgamma(k);
230
231 real log_F_T_d = gengamma_lcdf(d | shape, scale, k);
232 real log_tF_T_d = gengamma_lcdf(d | shape, scale, k_shift);
233
234 // q-dependent terms (guard only to avoid log(0); final algebra is unified).
235 real log_q_F_T_q; // log(q * F_T(q))
236 real log_E_tF_T_q; // log(E * tilde F_T(q))
237 if (q > 0) {
238 log_q_F_T_q = log(q) + gengamma_lcdf(q | shape, scale, k);
239 log_E_tF_T_q = log_E + gengamma_lcdf(q | shape, scale, k_shift);
240 } else {
241 log_q_F_T_q = negative_infinity();
242 log_E_tF_T_q = negative_infinity();
243 }
244
245 // Unified form: F_{S+}(d) = (A - B) / w_P with
246 // A = d * F_T(d) + E * tilde F_T(q)
247 // B = q * F_T(q) + E * tilde F_T(d)
248 // Ordering A >= B is guaranteed by F_{S+}(d) >= 0.
249 real log_A = log_sum_exp(log(d) + log_F_T_d, log_E_tF_T_q);
250 real log_B = log_sum_exp(log_q_F_T_q, log_E + log_tF_T_d);
251
252 return log_diff_exp(log_A, log_B) - log_window;
253}

260real primarycensored_analytical_lcdf_raw(data real d, int dist_id,
261 array[] real params,
262 data real pwindow,
263 int primary_id,
264 array[] real primary_params) {
265 real q = max({d - pwindow, 0});
266
267 if (dist_id == 2 && primary_id == 1) {
268 return primarycensored_gamma_uniform_lcdf(d | q, params, pwindow);
269 } else if (dist_id == 1 && primary_id == 1) {
270 return primarycensored_lognormal_uniform_lcdf(d | q, params, pwindow);
271 } else if (dist_id == 3 && primary_id == 1) {
272 return primarycensored_weibull_uniform_lcdf(d | q, params, pwindow);
273 } else if (dist_id == 5 && primary_id == 1) {
274 return primarycensored_gengamma_uniform_lcdf(d | q, params, pwindow);
275 } else if (dist_id == 26) {
276 // params = [boundaries (K+1), pmf (K)]; length 2*K + 1.
277 int K = (size(params) - 1) %/% 2;
278 return discretestep_lcdf(
279 d | to_vector(segment(params, 1, K + 1)),
280 to_vector(segment(params, K + 2, K)),
281 primary_id, primary_params, pwindow
282 );
283 } else if (dist_id == 27 || dist_id == 28) {
284 // params = [boundaries (K+1), hazards (K)]; length 2*K + 1. The last
285 // hazard must equal 1. RW (27) and RE (28) only differ in their
286 // prior so they share this likelihood dispatch.
287 int K = (size(params) - 1) %/% 2;
288 return discretehazard_lcdf(
289 d | to_vector(segment(params, 1, K + 1)),
290 to_vector(segment(params, K + 2, K)),
291 primary_id, primary_params, pwindow
292 );
293 }
294 return negative_infinity();
295}
